// VARIABEL ALAMAT SERVER STREAMING & METADATA BARU (FINAL) const STREAM_URL = "https://radioislam.my.id"; // Memanfaatkan Endpoint JSON resmi Shoutcast lewat jalur CorsProxy.io yang super cepat const METADATA_URL = "https://corsproxy.io/?" + encodeURIComponent("https://radioislam.my.id"); const audio = document.getElementById('audio-player'); const playBtn = document.getElementById('play-btn'); const statusText = document.getElementById('status-text'); const trackTitleDisplay = document.getElementById('track-title'); const helpBtn = document.getElementById('help-btn'); let isPlaying = false; // FUNGSI PENGAMBILAN METADATA JUDUL KAJIAN SECARA REAL-TIME async function fetchMetadata() { try { const response = await fetch(METADATA_URL); if (response.ok) { const data = await response.json(); // Membaca objek JSON bawaan Shoutcast (songtitle) if (data && data.songtitle) { trackTitleDisplay.innerText = data.songtitle; } else { trackTitleDisplay.innerText = "Siaran Kajian Islam Sragen"; } } } catch (error) { console.log("Gagal memuat judul otomatis, mencoba fallback...", error); // JALUR CADANGAN: Jika JSON dibatasi, tembak stats via proxy alternatif try { const fallbackResponse = await fetch("https://corsproxy.io/?" + encodeURIComponent("https://radioislam.my.id")); if (fallbackResponse.ok) { const textData = await fallbackResponse.text(); const statsArray = textData.split(','); if (statsArray.length >= 7 && statsArray[6].trim() !== "") { trackTitleDisplay.innerText = statsArray[6]; return; } } } catch (e) { console.log("Jalur cadangan juga gagal."); } trackTitleDisplay.innerText = "Kajian Islam Live (Aktif)"; } } // Jalankan sistem pembaca judul saat pertama kali dibuka, dan perbarui otomatis setiap 8 detik fetchMetadata(); setInterval(fetchMetadata, 8000); // FUNGSI UTAMA UNTUK MENYALAKAN AUDIO function startStreaming() { audio.src = STREAM_URL + "?ts=" + Date.now(); audio.play() .then(() => { isPlaying = true; playBtn.innerText = "HENTIKAN"; playBtn.style.background = "#b33939"; statusText.innerText = "Memutar Audio Streaming Live"; // Mekanisme menyalakan suara otomatis saat layar disentuh pendengar const unmuteAction = () => { audio.muted = false; statusText.innerText = "Memutar Audio Streaming Live"; window.removeEventListener('click', unmuteAction); window.removeEventListener('touchstart', unmuteAction); }; window.addEventListener('click', unmuteAction); window.addEventListener('touchstart', unmuteAction); }) .catch(err => { isPlaying = false; playBtn.innerText = "PUTAR"; playBtn.style.background = "#1b4d3e"; statusText.innerHTML = "Gagal memuat suara.
Diblokir oleh proteksi keamanan Browser."; helpBtn.style.display = "block"; }); } // Jalankan Autoplay Muted saat website selesai dimuat window.addEventListener('DOMContentLoaded', () => { startStreaming(); }); // Tombol manual kontrol kendali radio playBtn.addEventListener('click', (e) => { e.stopPropagation(); if (!isPlaying) { statusText.innerText = "Menghubungkan ke server..."; helpBtn.style.display = "none"; audio.muted = false; startStreaming(); } else { audio.pause(); audio.src = ""; isPlaying = false; playBtn.innerText = "PUTAR"; playBtn.style.background = "#1b4d3e"; statusText.innerText = "Streaming dihentikan"; helpBtn.style.display = "none"; } }); function showInstructions() { alert("Agar suara kajian bisa keluar, silakan:\n\n1. Klik ikon Pengaturan Situs / Gembok di sebelah kiri alamat URL web ini (bagian atas browser).\n2. Pilih 'Pengaturan Situs' (Site Settings).\n3. Cari menu 'Konten tidak aman' (Insecure content).\n4. Ubah opsinya menjadi 'Izinkan' (Allow).\n5. Buka kembali halaman ini dan segarkan (refresh)."); } // PWA Service Worker Registration if ('serviceWorker' in navigator) { window.addEventListener('load', () => { navigator.serviceWorker.register('sw.js') .then(reg => console.log('PWA Radio Islam Sragen Aktif!')) .catch(err => console.log('PWA Gagal:', err)); }); }
